%matplotlib inline
import matplotlib

    f.write(r.content)

#读取csv数据
train_data = pd.read_csv("kaggle_house_pred_train.csv")
test_data = pd.read_csv("kaggle_house_pred_test.csv")

#把去掉id的数据拼在一起，去掉id的目的是为了防止模型通过记住编号得到对应房价。
all_features = pd.concat((train_data.iloc[:, 1:-1], test_data.iloc[:, 1:]))

print("train_data.shape:", train_data.shape)
print("test_data.shape:", test_data.shape)
print("all_features:", all_features.shape)

test_data.shape: (1459, 80)

```

接下来我们需要对我们的数据进行处理，主要是分为数值型和非数值型。
```
#提取全是数字的特征名字
numeric_features = all_features.dtypes[all_features.dtypes != 'object'].index

#对数据做标准化处理,对应位置赋值
all_features[numeric_features] = all_features[numeric_features].apply(lambda x: (x - x.mean()) / (x.std()))

# 在标准化数据之后，将缺失值设置为0
all_features[numeric_features] = all_features[numeric_features].fillna(0)

#接下来将特征为类别值或者是离散值分成每一个特征为一个类别。
#dummy_na=true 我们将na视为有效的值
all_features = pd.get_dummies(all_features, dummy_na=True).astype(np.float32)

n_train = train_data.shape[0]

train_features = torch.tensor(all_features.iloc[:n_train].values, dtype=torch.float32)
test_features = torch.tensor(all_features.iloc[n_train:].values, dtype=torch.float32)
	train_labels = torch.tensor(np.log1p(train_data["SalePrice"].values).reshape(-1, 1), dtype=torch.float32)#需要将我们的结果变成log1p，因为某些数据太大了会导致我们的训练不稳定，会导致我们的数据被极端结果拉着走，loss会很大。

print("train_features.shape:", train_features.shape)
print("test_features.shape:", test_features.shape)
print("train_labels.shape:", train_labels.shape)
//train_features.shape: torch.Size([1460, 330])
test_features.shape: torch.Size([1459, 330])
train_labels.shape: torch.Size([1460, 1])

```

接下来将数据分批，变成迭代器。
```
# 数据分批
batch_size = 32
dataset = torch.utils.data.TensorDataset(train_features, train_labels)
train_loader = torch.utils.data.DataLoader(dataset, batch_size=batch_size, shuffle=True, num_workers=0)
print(f"每一批{len(next(iter(train_loader))[0])}个，一共{len(train_loader)}批")
