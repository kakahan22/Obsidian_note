class MyNet(torch.nn.Module):
    def __init__(self, input_dim, hidden1, hidden2, output_dim):
        super().__init__()
        self.linear1 = nn.Linear(input_dim, hidden1)
        self.linear2 = nn.Linear(hidden1, hidden2)
        self.linear3 = nn.Linear(hidden2, output_dim)
        self.relu = nn.ReLU()

    def forward(self, X):
        H1 = self.relu(self.linear1(X))
        H2 = self.relu(self.linear2(H1))
        H3 = self.linear3(H2)
        return H3

in_features = train_features.shape[1]
hidden1, hidden2, output_dim = 200, 100, 1
net = MyNet(in_features, hidden1, hidden2, output_dim).to(device)

"""
因为房价的值是几百万这种，会造成我们的损失函数特别大，所以我们更关心相对误差
解决这个方法，就是将我们的损失函数进行替换为对数函数
loss_fn = nn.MSELoss()"""

class LogRmSE(nn.Module):
    def __init__(self):
        super().__init__()
    def forward(self,pred,target):
        return torch.sqrt(torch.mean((torch.log1p(pred)-torch.log1p(target))**2))
log_fn = LogRmSE()
lr = 1e-2
wd = 0.0
optimizer = torch.optim.Adam(net.parameters(), lr=lr, weight_decay=wd)
print("in_features:", in_features)

接下来我们进行训练
```
epochs = 200

def train(train_loader):
    train_ls = []
    for epoch in range(epochs):
        loss_sum = 0.0
        for X, y in train_loader:
            X = X.to(device)
            y = y.to(device)

            optimizer.zero_grad()
            pred = net(X)
            l = log_fn(pred, y)
            l.backward()
            optimizer.step()
            loss_sum += l.item()
        train_ls.append(loss_sum)
        if (epoch + 1) % 20 == 0:
            print(f"epoch {epoch + 1}, loss = {loss_sum:.4f}")
    plt.plot(range(epochs), train_ls)
    plt.show()
    return train_ls

train(train_loader)
